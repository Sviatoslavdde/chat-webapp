        const LANGS = [
            { code: 'uk', flag: '🇺🇦', name: 'Українська' },
            { code: 'en', flag: '🇬🇧', name: 'English' },
            { code: 'de', flag: '🇩🇪', name: 'Deutsch' },

        ];

        let currentLang = 'uk';
        const userId = tg.initDataUnsafe?.user?.id;

        function init() {
            const tgLang = tg.initDataUnsafe?.user?.language_code || 'uk';
            if (I18N[tgLang]) currentLang = tgLang;

            if (userId && ADMIN_IDS.includes(userId)) {
                document.getElementById('tab-admin').style.display = 'block';
            }

            renderLangs();
            applyLanguage(currentLang);
        }

        function renderLangs() {
            const container = document.getElementById('lang-container');
            container.innerHTML = '';
            LANGS.forEach(l => {
                const item = document.createElement('div');
                item.className = `lang-item ${l.code === currentLang ? 'selected' : ''}`;
                item.innerHTML = `<span>${l.flag}</span> <span>${l.name}</span>`;
                item.onclick = () => selectLang(l.code);
                container.appendChild(item);
            });
        }

        function selectLang(code) {
            currentLang = code;
            renderLangs();
            applyLanguage(code);
            sendAction('set_lang', { lang: code });
        }

        function applyLanguage(lang) {
            const t = I18N[lang] || I18N.uk;
            document.getElementById('app-title').innerText = t.title;
            document.getElementById('app-subtitle').innerText = t.sub;
            document.getElementById('tab-main').innerText = t.t_main;
            document.getElementById('tab-settings').innerText = t.t_set;
            document.getElementById('tab-admin').innerText = t.t_adm;
            document.getElementById('status-title').innerText = t.st_title;
            document.getElementById('status-desc').innerText = t.st_idle;
            document.getElementById('btn-search').innerText = t.b_search;
            document.getElementById('btn-stop').innerText = t.b_stop;
            document.getElementById('btn-back').innerText = t.b_back;
            document.getElementById('lang-title').innerText = t.l_title;
        }

        function switchTab(tab) {
            document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
            document.querySelectorAll('.tab-content').forEach(c => c.classList.remove('active'));
            document.getElementById(`tab-${tab}`).classList.add('active');
            document.getElementById(`content-${tab}`).classList.add('active');
        }

        function sendAction(action, extra = {}) {
            const payload = JSON.stringify({ action: action, ...extra });
            tg.sendData(payload);
            tg.close();
        }

        init();
